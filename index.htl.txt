        <div class="level">Уровень: <span id="level">1</span> (XP: <span id="xp">0</span>/100)</div>
        
        <div id="cat" class="cat"></div>
        
        <div class="footer">Нажимай на Томми и зарабатывай!</div>

    <script>
        let tg = window.Telegram.WebApp;
        tg.expand();
        
        let balance = 0;
        let level = 1;
        let userId = tg.initDataUnsafe?.user?.id;

        // Загрузка профиля при старте
        async function loadProfile() {
            if (!userId) {
                document.getElementById('balance').innerText = "Ошибка (нет ID)";
                return;
            }
            try {
                let response = await fetch(`/api/profile?user_id=${userId}`);
                let data = await response.json();
                balance = data.balance;
                level = data.level;
                updateUI(data.balance, data.level, data.xp);
            } catch (e) { console.log("Ошибка загрузки", e); }
        }

        function updateUI(bal, lvl, xp) {
            document.getElementById('balance').innerText = bal.toFixed(2);
            document.getElementById('level').innerText = lvl;
            document.getElementById('xp').innerText = xp;
        }

        // Обработка клика
        document.getElementById('cat').addEventListener('click', async () => {
            tg.HapticFeedback.impactOccurred('light');
            
            // Мгновенное обновление для отзывчивости
            let currentBal = parseFloat(document.getElementById('balance').innerText);
            document.getElementById('balance').innerText = (currentBal + 0.01).toFixed(2);

            if (userId) {
                try {
                    let response = await fetch('/api/click', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ user_id: userId })
                    });
                    let data = await response.json();
                    updateUI(data.balance, data.level, data.xp);
                } catch (e) { console.log("Ошибка клика", e); }
            }
        });

        loadProfile();
    </script>
